cmake_minimum_required(VERSION 3.22)
project(PeaceDukeBotTests LANGUAGES CXX)
enable_testing()
find_package(Threads REQUIRED)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(PEACE_COVERAGE "Instrument original application code" OFF)
set(PEACE_SANITIZER "" CACHE STRING "address, undefined or thread")
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
file(GLOB MODULES CONFIGURE_DEPENDS "${ROOT}/src/modules/*.cpp" "${ROOT}/src/modules/commands/*.cpp")
add_library(app_under_test STATIC ${MODULES} "${ROOT}/src/bot_core.cpp" "${ROOT}/src/engine/console.cpp" "${ROOT}/src/engine/json_database.cpp")
target_include_directories(app_under_test PUBLIC mocks vendor "${ROOT}/src")
# Test-only compiler switch exposes private helpers without altering any header or class layout.
target_compile_options(app_under_test PUBLIC -fno-access-control)
target_link_libraries(app_under_test PUBLIC Threads::Threads)
if(PEACE_COVERAGE)
 target_compile_options(app_under_test PRIVATE --coverage -O0 -g)
 target_link_options(app_under_test PUBLIC --coverage)
endif()
if(PEACE_SANITIZER)
 add_compile_options(-fsanitize=${PEACE_SANITIZER} -fno-omit-frame-pointer -g)
 add_link_options(-fsanitize=${PEACE_SANITIZER})
 target_compile_options(app_under_test PRIVATE -fsanitize=${PEACE_SANITIZER} -fno-omit-frame-pointer -g)
 target_link_options(app_under_test PUBLIC -fsanitize=${PEACE_SANITIZER})
endif()
add_executable(peace_tests test_main.cpp engine_tests.cpp module_tests.cpp regression_tests.cpp)
target_link_libraries(peace_tests PRIVATE app_under_test)
# Let CTest and sanitizers report fatal signals directly.
target_compile_definitions(peace_tests PRIVATE DOCTEST_CONFIG_NO_POSIX_SIGNALS)
add_executable(legacy_converter "${ROOT}/src/tools/parse_old_db.cpp")
target_link_libraries(legacy_converter PRIVATE app_under_test)
# Discover each case as a separate process: fatal regressions cannot hide subsequent cases.
add_custom_command(TARGET peace_tests POST_BUILD COMMAND ${CMAKE_COMMAND}
 -DTEST_EXECUTABLE=$<TARGET_FILE:peace_tests> -DTEST_FILE=${CMAKE_CURRENT_BINARY_DIR}/cases.cmake
 -P ${CMAKE_CURRENT_SOURCE_DIR}/discover.cmake)
set_property(DIRECTORY APPEND PROPERTY TEST_INCLUDE_FILES "${CMAKE_CURRENT_BINARY_DIR}/cases.cmake")
add_executable(bot_cli "${ROOT}/src/main.cpp")
target_link_libraries(bot_cli PRIVATE app_under_test)
if(PEACE_COVERAGE)
 foreach(t peace_tests legacy_converter bot_cli)
  target_compile_options(${t} PRIVATE --coverage -O0 -g)
 endforeach()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(legacy_db_stress "${ROOT}/src/tests/db.cpp")
target_link_libraries(legacy_db_stress PRIVATE app_under_test)
foreach(kind legacy_db_stress main_missing_token main_offline_start converter_missing_args converter_roundtrip converter_invalid_age)
 if(kind MATCHES "^main")
  set(binary bot_cli)
 elseif(kind STREQUAL "legacy_db_stress")
  set(binary legacy_db_stress)
 else()
  set(binary legacy_converter)
 endif()
 add_test(NAME "integration.cli.${kind}" COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/cli_tests.py" "${kind}" "$<TARGET_FILE:${binary}>")
 set_tests_properties("integration.cli.${kind}" PROPERTIES TIMEOUT 15)
endforeach()
